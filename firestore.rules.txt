rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }
    function isSuperAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/superadmins/$(request.auth.uid));
    }
    function myUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function myShopId() { return myUserData().shopId; }
    function myRole() { return myUserData().role; }
    function isAgent() { return isSignedIn() && myRole() == 'agent'; }
    function myAgentMarketId() {
      return get(/databases/$(database)/documents/agents/$(request.auth.uid)).data.marketId;
    }

    match /superadmins/{uid} {
      allow read: if isSignedIn();
      allow write: if isSuperAdmin();
    }

    match /system/{docId} {
      allow read: if isSignedIn();
      allow write: if isSuperAdmin();
    }

    match /users/{uid} {
      allow create: if isSignedIn() &&
        (request.auth.uid == uid || isSuperAdmin() || isAgent() ||
         (myRole() == 'admin' && request.resource.data.shopId == myShopId()));
      allow read: if isSignedIn() &&
        (request.auth.uid == uid || isSuperAdmin() ||
         myShopId() == get(/databases/$(database)/documents/users/$(uid)).data.shopId);
      allow update: if isSignedIn() && (request.auth.uid == uid || isSuperAdmin());
      allow delete: if isSuperAdmin();
    }

    match /shops/{shopId} {
      allow create: if isSignedIn() && request.auth.uid == shopId;
      allow get: if true;
      allow list: if true;
      allow update: if isSignedIn() &&
        (isSuperAdmin() ||
         (myShopId() == shopId && myRole() == 'admin') ||
         (resource.data.assignedAgentId == request.auth.uid));
      allow delete: if isSuperAdmin();

      match /members/{memberUid} {
        allow read: if isSignedIn() && (isSuperAdmin() || myShopId() == shopId);
        allow write: if isSignedIn() &&
          (isSuperAdmin() || (myShopId() == shopId && myRole() == 'admin'));
      }

      match /appdata/{docId} {
        allow read: if isSignedIn() && (isSuperAdmin() || myShopId() == shopId);
        allow write: if isSignedIn() && myShopId() == shopId;
      }

      match /deliveryCalls/{callId} {
        allow read: if true;
        allow create: if isSignedIn() && myShopId() == shopId;
        allow update: if true;
      }

      match /deliveryDevices/{token} {
        allow read, write: if true;
      }

      match /publicStock/{itemId} {
        allow read: if true;
        allow write: if isSignedIn() && myShopId() == shopId;
      }

      match /publicCategories/{catId} {
        allow read: if true;
        allow write: if isSignedIn() && myShopId() == shopId;
      }

      match /live/{docId} {
        allow read, write: if true;
      }
    }

    match /markets/{marketId} {
      allow read: if true;
      allow write: if isSuperAdmin();
    }

    match /agents/{agentId} {
      allow get: if true;
      allow list: if true;
      allow create: if isSuperAdmin();
      allow update: if isSignedIn() && (isSuperAdmin() || request.auth.uid == agentId);
      allow delete: if isSuperAdmin();
    }

    match /riders/{riderId} {
      allow read: if isSignedIn() &&
        (isSuperAdmin() || resource.data.assignedAgentId == request.auth.uid || request.auth.uid == riderId);
      allow create: if isSignedIn() && request.resource.data.assignedAgentId == request.auth.uid;
      allow update: if isSignedIn() &&
        (isSuperAdmin() || resource.data.assignedAgentId == request.auth.uid || request.auth.uid == riderId);
      allow delete: if isSuperAdmin();
    }

    match /transportDrivers/{id} {
      allow get: if true;
      allow list: if true;
      allow create: if isSignedIn() && request.auth.uid == id;
      allow update: if isSignedIn() &&
        (isSuperAdmin() || request.auth.uid == id ||
         (isAgent() && resource.data.marketId == myAgentMarketId()));
      allow delete: if isSuperAdmin();
    }

    match /trips/{tripId} {
      allow read: if true;
      allow create: if true;
      allow update: if true;
    }

    match /tripOffers/{offerId} {
      allow read: if true;
      allow update: if true;
      allow create, delete: if false;
    }

    match /deliveryOffers/{offerId} {
      allow read: if isSignedIn() && (isSuperAdmin() || resource.data.riderId == request.auth.uid);
      allow update: if isSignedIn() && resource.data.riderId == request.auth.uid;
      allow create, delete: if false;
    }

    match /orderRequests/{orderId} {
      allow create: if isSignedIn();
      allow read: if isSignedIn();
      allow update: if isSignedIn();
    }

    match /passenger_waiting_queue/{docId} {
      allow read, create, update, delete: if true;
    }

    match /customers/{uid} {
      allow read, write: if isSignedIn() && request.auth.uid == uid;
    }

    match /wallets/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isSuperAdmin() || isAgent());
      allow write: if isSignedIn();

      match /transactions/{txId} {
        allow read: if isSignedIn() && (request.auth.uid == uid || isSuperAdmin() || isAgent());
        allow write: if isSignedIn();
      }
    }

    match /bloodDonors/{uid} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.uid == uid;

      match /comments/{commentId} {
        allow read: if true;
        allow create: if isSignedIn();
        allow update, delete: if isSignedIn() && resource.data.uid == request.auth.uid;
      }
    }

    match /bloodRequests/{requestId} {
      allow read: if true;
      allow create: if isSignedIn() && request.resource.data.requesterUid == request.auth.uid;
      allow update: if isSignedIn() && resource.data.requesterUid == request.auth.uid;

      match /comments/{commentId} {
        allow read: if true;
        allow create: if isSignedIn();
        allow update, delete: if isSignedIn() && resource.data.uid == request.auth.uid;
      }
    }

    match /complaints/{complaintId} {
      allow read: if true;
      allow create: if isSignedIn() && request.resource.data.complainantUid == request.auth.uid;
      allow update: if isSuperAdmin() || isAgent();
    }

    match /messengerUsers/{uid} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && request.auth.uid == uid;
    }

    match /messengerUsernames/{username} {
      allow read: if true;
      allow write: if isSignedIn();
    }

    match /messengerFriendRequests/{pairId} {
      allow read: if isSignedIn() && (request.auth.uid == resource.data.fromUid || request.auth.uid == resource.data.toUid);
      allow create: if isSignedIn() && request.auth.uid == request.resource.data.fromUid;
      allow update: if isSignedIn() && (request.auth.uid == resource.data.fromUid || request.auth.uid == resource.data.toUid);
    }

    match /messengerChats/{chatId} {
      allow read: if isSignedIn() && request.auth.uid in resource.data.participants;
      allow create: if isSignedIn() && request.auth.uid in request.resource.data.participants;
      allow update: if isSignedIn() && request.auth.uid in resource.data.participants;

      match /messages/{messageId} {
        allow read: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.participants;
        allow create: if isSignedIn() && request.resource.data.senderUid == request.auth.uid;
        allow update: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.participants;
        allow delete: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/messengerChats/$(chatId)).data.participants;
      }
    }

    match /ambulance_call_logs/{logId} {
      allow read: if isSignedIn() && (isSuperAdmin() || isAgent() || resource.data.customerGoogleId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.customerGoogleId == request.auth.uid;
      allow update: if isSignedIn() && resource.data.customerGoogleId == request.auth.uid;
    }

    match /medicalEquipment/{equipmentId} {
      allow read: if true;
      allow write: if isSignedIn();
    }

    match /medicalEquipmentOrders/{orderId} {
      allow read: if isSignedIn() && (isSuperAdmin() || isAgent() || resource.data.customerGoogleId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.customerGoogleId == request.auth.uid;
      allow update: if isSignedIn();
    }

    match /activeSessions/{sessionId} {
      allow create, update: if true;
      allow read: if isSuperAdmin();
      allow delete: if isSuperAdmin();
    }

    match /loginAttempts/{email} {
      allow read: if true;
      allow delete: if true;
      allow create, update: if request.resource.data.keys().hasOnly(['count','lastAttempt']) &&
        request.resource.data.count is int && request.resource.data.count <= 20;
    }

    match /imageIndex/{imageId} {

      allow read: if isSignedIn() && myShopId() != null;

      allow create: if isSignedIn()
        && myShopId() != null
        && request.resource.data.imageId == imageId
        && request.resource.data.referenceCount == 0
        && request.resource.data.uploaderShopId == myShopId()
        && request.resource.data.format == 'webp'
        && request.resource.data.sizeBytes <= 200 * 1024
        && request.resource.data.keys().hasAll(
          ['imageId','sha256','storagePath','imageUrl','sizeBytes',
           'format','referenceCount','cleanupStatus','orphanedAt',
           'uploaderShopId','createdAt','updatedAt']);

      allow update: if isSignedIn()
        && myShopId() != null
        && request.resource.data.imageId == resource.data.imageId
        && request.resource.data.sha256 == resource.data.sha256
        && request.resource.data.storagePath == resource.data.storagePath
        && request.resource.data.imageUrl == resource.data.imageUrl
        && request.resource.data.sizeBytes == resource.data.sizeBytes
        && request.resource.data.format == resource.data.format
        && request.resource.data.uploaderShopId == resource.data.uploaderShopId
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.referenceCount >= 0
        && (
          request.resource.data.referenceCount ==
            resource.data.referenceCount + 1
          ||
          request.resource.data.referenceCount ==
            resource.data.referenceCount - 1
        );

      allow delete: if false;
    }
  }
}
